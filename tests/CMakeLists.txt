# Smoke tests for POM1. Enabled by default on native builds; skipped for
# Emscripten (ctest runs the executable on the host, not in the browser).

# assert() must stay LIVE in every test. The top-level Release build compiles
# with -DNDEBUG (CMAKE_CXX_FLAGS_RELEASE), which silently turned every
# <cassert>-based pin in this directory into a no-op — the binaries ran and
# "passed" while checking nothing (discovered juillet 2026 via the convert
# smoke tests). Directory-level compile options are emitted AFTER the
# per-config flags on the compiler command line, so the undef wins for the
# test translation units only; the emulator itself keeps its NDEBUG Release
# build. (Tests using custom must()/mustBeTrue() helpers were never affected.)
if(MSVC)
    add_compile_options(/UNDEBUG)
else()
    add_compile_options(-UNDEBUG)
endif()

# Same warning flags as the emulator (set in the top-level CMakeLists). Nothing
# vendored is compiled in this directory, so unlike the main target these can be
# applied directory-wide rather than per source file.
add_compile_options(${POM1_WARNING_FLAGS})

# ...with one exception, and it is not "nothing vendored is compiled here" after
# all: AudioDevice.cpp #includes the vendored stb_vorbis.c textually, and that
# trips C4701. MSVC decides 4700+ warnings during code generation, so no pragma
# in the source can suppress them — see the twin block in the top-level
# CMakeLists.txt, repeated here because source properties are directory-scoped.
if(MSVC)
    set_source_files_properties(${CMAKE_SOURCE_DIR}/src/AudioDevice.cpp
                                PROPERTIES COMPILE_OPTIONS "/wd4701")
endif()

# Tests compiling AudioDevice.cpp (miniaudio, via POM1_TEST_CORE_SOURCES) need
# libdl + libpthread. The main app gets both transitively from the glfw target;
# tests don't link glfw. Modern glibc (>= 2.34) folds them into libc so the
# omission was invisible — bionic's glibc 2.27 (the AppImage release container,
# run 29919749167) links them as separate libraries. Directory-scoped so every
# test target defined below picks them up.
if(UNIX)
    find_package(Threads REQUIRED)
    link_libraries(Threads::Threads ${CMAKE_DL_LIBS})
endif()

# Portability macros, directory-scoped so pom1_test_devices and every test target below
# compiles POM1's sources under the SAME macro set as the app. Defined as an
# INTERFACE target in the top-level CMakeLists; see the long note there for why
# a per-target copy is not good enough (C2589 on RewindBuffer.cpp, 22 août
# 2026: the app built, three test binaries did not).
link_libraries(pom1_build_flags)

# Apple deprecated the whole OpenGL API in 10.14. The peripheral .cpp files
# transitively include it, so every test target needs the silencer — it used to
# be a four-line `if(APPLE)` block repeated verbatim 50 times below, once per
# target. Directory-scoped, it is stated once and a new test target cannot
# forget it. (The app's own copy stays per-target and conditional: there it is
# deliberately applied only on the legacy POM1_RENDERER=opengl macOS path.)
if(APPLE)
    add_compile_definitions(GL_SILENCE_DEPRECATION)
endif()

# -- Klaus Dormann 6502 functional test ---------------------------------------
# Downloads the pre-assembled test binary once at configure time, builds a
# minimal runner that links against POM1's M6502 + Memory + peripheral core,
# and registers it with ctest.

set(POM1_KLAUS_BIN "${CMAKE_CURRENT_BINARY_DIR}/klaus_6502_functional_test.bin"
    CACHE FILEPATH "Path to Klaus Dormann's 6502 functional-test binary")
get_filename_component(KLAUS_BIN "${POM1_KLAUS_BIN}" ABSOLUTE
                       BASE_DIR "${CMAKE_BINARY_DIR}")
set(KLAUS_URL "https://raw.githubusercontent.com/Klaus2m5/6502_65C02_functional_tests/master/bin_files/6502_functional_test.bin")
set(KLAUS_SHA256 "fa12bfc761e6f9057e4cc01a665a7b800ff01ae91f598af1e39a1201d01953fd")
set(KLAUS_AVAILABLE FALSE)
set(KLAUS_DOWNLOADED FALSE)

if(NOT EXISTS "${KLAUS_BIN}")
    message(STATUS "Downloading Klaus 6502 functional test binary (~64 KB)...")
    get_filename_component(KLAUS_BIN_DIR "${KLAUS_BIN}" DIRECTORY)
    file(MAKE_DIRECTORY "${KLAUS_BIN_DIR}")
    file(DOWNLOAD "${KLAUS_URL}" "${KLAUS_BIN}"
         TIMEOUT 30
         STATUS DOWNLOAD_STATUS)
    list(GET DOWNLOAD_STATUS 0 DOWNLOAD_CODE)
    if(NOT DOWNLOAD_CODE EQUAL 0)
        list(GET DOWNLOAD_STATUS 1 DOWNLOAD_MSG)
        message(WARNING "Klaus test binary download failed: ${DOWNLOAD_MSG} — "
                        "klaus_6502_functional will not be registered. "
                        "Provide it with -DPOM1_KLAUS_BIN=/path/to/6502_functional_test.bin")
        file(REMOVE "${KLAUS_BIN}")  # leave no truncated file behind
    else()
        set(KLAUS_DOWNLOADED TRUE)
    endif()
endif()

# Hash only a complete local file. CMake's EXPECTED_HASH makes a network error
# look like a checksum error and can abort configuration before the test can be
# disabled cleanly.
if(EXISTS "${KLAUS_BIN}")
    file(SHA256 "${KLAUS_BIN}" KLAUS_ACTUAL_SHA256)
    if(KLAUS_ACTUAL_SHA256 STREQUAL KLAUS_SHA256)
        set(KLAUS_AVAILABLE TRUE)
    else()
        message(WARNING
            "Klaus binary at ${KLAUS_BIN} has SHA-256 ${KLAUS_ACTUAL_SHA256}; "
            "expected ${KLAUS_SHA256}. klaus_6502_functional will not be registered.")
        if(KLAUS_DOWNLOADED)
            file(REMOVE "${KLAUS_BIN}")
        endif()
    endif()
endif()

# Common wiring for tests that exercise the shared device object set. Optional
# ARGS are appended after the executable and TIMEOUT defaults to 30 seconds.
function(pom1_add_smoke_test TARGET SOURCE TEST_NAME)
    cmake_parse_arguments(SMOKE "" "TIMEOUT" "ARGS" ${ARGN})
    if(NOT SMOKE_TIMEOUT)
        set(SMOKE_TIMEOUT 30)
    endif()
    add_executable(${TARGET} ${SOURCE} ${POM1_TEST_CORE_SOURCES})
    target_link_libraries(${TARGET} PRIVATE pom1_devices)
    add_test(NAME ${TEST_NAME} COMMAND ${TARGET} ${SMOKE_ARGS})
    set_tests_properties(${TEST_NAME} PROPERTIES TIMEOUT ${SMOKE_TIMEOUT})
endfunction()

# so a shared CI runner can blow its bounds without anything being wrong. Seen
# on GitHub's Linux runner in septembre 2026: state-hold 279 ms against a 100 ms
# gate and state-wait 9.3 s against a 5 s one, on a commit that touched nothing
# in the lock path, with the same commit passing on the next run and `main`
# green twenty minutes earlier.
#
# The retry is the honest fix rather than raising the bounds. The bounds are
# what the test is FOR — 100 ms is the point at which a held stateMutex parks
# the audio callback and the render thread, and CLAUDE.md records that it caught
# exactly that when the SID chip was first built inside the lock. Relaxing them
# to survive a noisy runner would keep the test green and stop it meaning
# anything; three independent failures still fail the job, and a real regression
# does not pass one run in three.
#
# It matters beyond CI hygiene: a newcomer whose first pull request goes red on
# an unrelated flake concludes they broke something. That is a contribution
# barrier, so it is documented in CONTRIBUTING.md too.
set_tests_properties(concurrent_frontends_smoke PROPERTIES
    TIMEOUT 30
    REPEAT UNTIL_PASS:3
    LABELS "concurrency;audio")

# ── Bind shared-object consumers to their logical layer ─────────────────────
# POM1_TEST_CORE_SOURCES is a generator expression in add_executable(), which
# contributes object files but does NOT propagate pom1_test_devices' usage
# requirements. Discover every consumer after all targets have been declared
# and bind it to pom1_devices. This closes the historical gap where adding a
# smoke test meant remembering src/, ImGui headers, libresidfp and ws2_32 by
# hand. A future consumer is classified automatically at configure time.
get_property(pom1_test_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
set(pom1_device_consumer_count 0)
foreach(pom1_target IN LISTS pom1_test_targets)
    get_target_property(pom1_target_type ${pom1_target} TYPE)
    if(NOT pom1_target_type STREQUAL "EXECUTABLE")
        continue()
    endif()
    get_target_property(pom1_target_sources ${pom1_target} SOURCES)
    if("$<TARGET_OBJECTS:pom1_test_devices>" IN_LIST pom1_target_sources)
        get_target_property(pom1_target_links ${pom1_target} LINK_LIBRARIES)
        if(pom1_target_links AND NOT pom1_target_links STREQUAL "pom1_target_links-NOTFOUND")
            # These requirements now arrive through pom1_devices. Remove the
            # legacy copies so Apple ld does not warn about duplicate archives
            # and Windows does not carry two independent ws2_32 declarations.
            list(REMOVE_ITEM pom1_target_links libresidfp_static ws2_32)
            set_property(TARGET ${pom1_target} PROPERTY LINK_LIBRARIES
                         "${pom1_target_links}")
        endif()
        if(NOT "pom1_devices" IN_LIST pom1_target_links)
            target_link_libraries(${pom1_target} PRIVATE pom1_devices)
        endif()
        math(EXPR pom1_device_consumer_count "${pom1_device_consumer_count} + 1")

        # Configuration-time assertion: the source/object relationship and
        # declared architecture layer may never drift apart again.
        get_target_property(pom1_verified_links ${pom1_target} LINK_LIBRARIES)
        if(NOT "pom1_devices" IN_LIST pom1_verified_links)
            message(FATAL_ERROR
                "${pom1_target} consumes pom1_test_devices but is not linked to pom1_devices")
        endif()
    endif()
endforeach()
if(pom1_device_consumer_count EQUAL 0)
    message(FATAL_ERROR "No pom1_test_devices consumer was classified")
endif()
message(STATUS
    "POM1: ${pom1_device_consumer_count} test targets bound to pom1_devices")

# Every TIMEOUT above is sized for an optimised build. ASan/UBSan/TSan slow the
# emulated 6502 down by several times, and POM1's slowest tests boot a whole
# BASIC interpreter on it — applesoft_gen2_smoke and applesoft_tms9918_smoke
# both blew their 120 s budget on the first instrumented run here, which reads
# in CI as "the sanitizer job is red" when in fact it found nothing at all.
#
# A false red is worse than no job: it trains you to ignore the one report that
# was supposed to be believed. So scale every declared timeout instead of
# picking new numbers by hand in ninety places — the ratios between tests are
# already right, only the clock changed.
#
# The factor is MEASURED, not guessed, and it is far bigger than the usual
# 2-4x rule of thumb: ASan instruments every access to Memory's 64 KB array,
# which IS the 6502 interpreter's innermost loop. The emulated-CPU-bound tests
# slow down by at least 20x (applesoft_gen2_smoke: 31.6 s clean, still running
# past 600 s instrumented). 30x leaves real headroom on a slower CI runner, and
# costs nothing on the ~80 tests that finish under a second either way.
if(POM1_SANITIZE)
    get_property(pom1_declared_tests DIRECTORY PROPERTY TESTS)
    foreach(pom1_t IN LISTS pom1_declared_tests)
        get_test_property(${pom1_t} TIMEOUT pom1_tmo)
        if(pom1_tmo AND NOT pom1_tmo STREQUAL "NOTFOUND")
            math(EXPR pom1_scaled "${pom1_tmo} * 30")
            set_tests_properties(${pom1_t} PROPERTIES TIMEOUT ${pom1_scaled})
        endif()
    endforeach()
    message(STATUS "POM1: test timeouts scaled x30 for the sanitizer build")
endif()

# ── Applying the lanes ───────────────────────────────────────────────────────
# Run last: DIRECTORY property TESTS only holds what has been declared so far.
# A name listed above but not declared in this configuration is NOT an error —
# the cc65 tests live inside if(NOT WIN32) and size_report_smoke inside
# if(Python3_FOUND) — so it is reported rather than fatal. A typo shows up in
# that same report, and shows up again as an emulator lane that needs cc65.
get_property(pom1_declared_lane_tests DIRECTORY PROPERTY TESTS)
set(pom1_lane_missing "")
foreach(pom1_lane_t IN LISTS POM1_DEVTOOLS_TESTS)
    if(NOT pom1_lane_t IN_LIST pom1_declared_lane_tests)
        list(APPEND pom1_lane_missing ${pom1_lane_t})
    endif()
endforeach()
set(pom1_lane_emulator 0)
set(pom1_lane_devtools 0)
foreach(pom1_lane_t IN LISTS pom1_declared_lane_tests)
    if(pom1_lane_t IN_LIST POM1_DEVTOOLS_TESTS)
        set_property(TEST ${pom1_lane_t} APPEND PROPERTY LABELS devtools)
        math(EXPR pom1_lane_devtools "${pom1_lane_devtools} + 1")
    else()
        set_property(TEST ${pom1_lane_t} APPEND PROPERTY LABELS emulator)
        math(EXPR pom1_lane_emulator "${pom1_lane_emulator} + 1")
    endif()
endforeach()
message(STATUS
    "POM1: test lanes — ${pom1_lane_emulator} emulator, ${pom1_lane_devtools} devtools")
if(pom1_lane_missing)
    message(STATUS
        "POM1: devtools tests not declared in this configuration: ${pom1_lane_missing}")
endif()
